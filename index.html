async function sendMessage() { const input = document.getElementById('user-input'); const val = input.value.trim(); if(!val) return; addMsg(val, 'user-msg'); input.value = ""; const light = document.getElementById('status-light'); light.style.background = "#ffff00"; // Mode calcul try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${NDX_CONFIG.apiKey}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: `Tu es NDX.A, l'expert suprême universel créé par Ndiaye Adama. Ton ton est froid, précis et loyal. Tu maîtrises le code, les maths et le business. Réponds à : ${val}` }] }] }) }); const data = await response.json(); if (data.candidates && data.candidates[0].content.parts[0].text) { const aiText = data.candidates[0].content.parts[0].text; addMsg(aiText, 'ai-msg'); light.style.background = "#00d2ff"; // Succès } else { throw new Error("Réponse vide"); } } catch (e) { console.error("Erreur de noyau:", e); addMsg("[ERREUR SYSTÈME] : Clé API invalide ou quota dépassé. Mode secours activé.", 'ai-msg'); light.style.background = "#ff4d4d"; // Erreur } }
